{%- comment -%} Reusable journal cover graphic. Generates a deterministic SVG from the entry's category + title — no per-article image upload needed. Usage: {% include journal-cover.html entry=entry variant="thumb" %} {% include journal-cover.html entry=entry variant="banner" %} - category "Special" -> ochre palette, "Daily" (default) -> green palette - a hash of the title picks one of several glyph motifs so entries look distinct from one another. {%- endcomment -%} {%- assign e = include.entry -%} {%- assign cat = e.category | default: "Daily" | downcase -%} {%- comment -%} Derive a small stable number from the title for variety {%- endcomment -%} {%- assign seed = e.title | default: e.slug | size -%} {%- assign titlechars = e.title | default: "x" | split: "" -%} {%- assign acc = 0 -%} {%- for c in titlechars -%} {%- assign acc = acc | plus: forloop.index -%} {%- endfor -%} {%- assign motif = acc | modulo: 4 -%} {%- if cat == "special" -%} {%- assign c1 = "#b8862f" -%}{%- assign c2 = "#8f6314" -%}{%- assign cbg1 = "#faf4e6" -%}{%- assign cbg2 = "#f0e4c8" -%} {%- else -%} {%- assign c1 = "#3f7d6e" -%}{%- assign c2 = "#2f6657" -%}{%- assign cbg1 = "#eaf2ee" -%}{%- assign cbg2 = "#d8e7e0" -%} {%- endif -%}
